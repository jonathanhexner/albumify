The contract type — DashboardMetrics

@dataclass(frozen=True, slots=True)
class DashboardMetrics:
    total_runs:        int
    total_faces_ever:  int
    avg_n_clusters:    Optional[float]
    median_n_clusters: Optional[float]
    last_run_at:       Optional[str]      # raw iso — tab computes "age"
    gate_pass_rate:    Optional[float]    # fraction status=="complete"
    per_album:         List[AlbumStat]    # {album, n_runs}  desc
    per_profile:       List[ProfileStat]  # {profile, n_runs} desc
    timeseries:        List[RunPoint]     # {run_id, started_at, n_clusters, album}
